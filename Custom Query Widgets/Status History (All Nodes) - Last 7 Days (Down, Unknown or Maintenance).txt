select

    n.caption as [Device]
    -- shows the current status icon
    ,'/Orion/images/StatusIcons/Small-' + n.StatusIcon AS [_IconFor_Device]
    -- makes a clickable link to the node details
    ,n.DetailsUrl as [_linkfor_Device]
    ,TOSTRING(TOLOCAL(history.DateTimeFrom)) AS [From]
    ,ISNULL(TOSTRING(TOLOCAL(history.DateTimeUntil)),'Still') AS [Till]
    ,s.StatusName AS [Status]
    ,'/Orion/images/StatusIcons/Small-' + s.StatusName + '.gif' AS [_IconFor_Status]
    ,history.TotalDurationMin AS [Min]

FROM Orion.NetObjectDowntime history

JOIN Orion.Nodes n ON history.NodeID = n.NodeID
JOIN Orion.StatusInfo s ON history.State = s.StatusId

WHERE

    history.EntityType = 'Orion.Nodes' and
     history.TotalDurationMin > 0 and 
        (history.State = 2 OR history.State = 0 OR history.State = 9) -- Down, Unknown or Maintenance
    AND (DAYDIFF(history.DateTimeFrom, GETUTCDATE()) <= 7 OR DAYDIFF(history.DateTimeUntil, GETUTCDATE()) <= 7)

--AND s.StatusName like '%${SEARCH_STRING}%'


ORDER BY history.DateTimeFrom DESC
